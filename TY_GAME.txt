# 3. Develop Snake Game using Pygame 

import pygame,random,sys
pygame.init()
w,h=720,480
s=pygame.display.set_mode((w,h))
c=pygame.time.Clock()
snake=[[100,50],[90,50],[80,50]]
dir='RIGHT'
f=[random.randrange(0,w//10)*10,random.randrange(0,h//10)*10]
score=0

while True:
    c.tick(15)
    for e in pygame.event.get():
        if e.type==pygame.QUIT: pygame.quit();sys.exit()
        if e.type==pygame.KEYDOWN:
            if e.key==pygame.K_UP and dir!='DOWN': dir='UP'
            if e.key==pygame.K_DOWN and dir!='UP': dir='DOWN'
            if e.key==pygame.K_LEFT and dir!='RIGHT': dir='LEFT'
            if e.key==pygame.K_RIGHT and dir!='LEFT': dir='RIGHT'

    head=snake[0][:]
    if dir=='UP': head[1]-=10
    if dir=='DOWN': head[1]+=10
    if dir=='LEFT': head[0]-=10
    if dir=='RIGHT': head[0]+=10
    snake.insert(0,head)
    if head==f: score+=10; f=[random.randrange(0,w//10)*10,random.randrange(0,h//10)*10]
    else: snake.pop()
    if head[0]<0 or head[0]>=w or head[1]<0 or head[1]>=h or head in snake[1:]: pygame.quit();sys.exit()
    s.fill((0,0,0))
    [pygame.draw.rect(s,(0,255,0),pygame.Rect(p[0],p[1],10,10)) for p in snake]
    pygame.draw.rect(s,(255,0,0),pygame.Rect(f[0],f[1],10,10))
    s.blit(pygame.font.SysFont(None,32).render(f"Score:{score}",True,(255,255,255)),(10,10))
    pygame.display.flip()

# 4. Create 2D Target Shooting Game 

import pygame,random,math,sys
pygame.init()
w,h=800,600
s=pygame.display.set_mode((w,h))
c=pygame.time.Clock()
p=pygame.Rect(370,523,64,32)
inv=[pygame.Rect(random.randint(64,736),random.randint(30,180),40,30) for _ in range(8)]
spd=[4]*8
b=pygame.Rect(0,0,10,25)
fired=False;score=0;go=False;ps=0
fnt=pygame.font.SysFont(None,32)

def hit(a,b): return math.hypot(a.centerx-b.centerx,a.y-b.y)<40

while True:
    c.tick(60)
    for e in pygame.event.get():
        if e.type==pygame.QUIT: pygame.quit();sys.exit()
        if e.type==pygame.KEYDOWN:
            if e.key==pygame.K_LEFT: ps=-5
            if e.key==pygame.K_RIGHT: ps=5
            if e.key==pygame.K_SPACE and not fired and not go: b.midbottom,fired=p.midtop,True
        if e.type==pygame.KEYUP and e.key in (pygame.K_LEFT,pygame.K_RIGHT): ps=0
    p.x=max(0,min(p.x+ps,736))
    if not go:
        for i,invader in enumerate(inv):
            invader.x+=spd[i]
            if invader.left<=0 or invader.right>=800: spd[i]*=-1; invader.y+=30
            if invader.y>=500: go=True
            if fired and hit(b,invader): score+=1; fired=False; invader.update(random.randint(64,736),random.randint(30,180),40,30)
    if fired: b.y-=12
    if b.y<=0: fired=False
    s.fill((0,0,0))
    pygame.draw.rect(s,(0,255,255),p)
    [pygame.draw.rect(s,(0,255,0),i) for i in inv]
    if fired: pygame.draw.rect(s,(255,255,0),b)
    s.blit(fnt.render(f"Score:{score}",True,(255,255,255)),(10,10))
    pygame.display.flip()
