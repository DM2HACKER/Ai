# 1. Columnar Transposition


def columnar_encrypt(plaintext, key):
    # Remove spaces and convert to uppercase
    plaintext = plaintext.replace(' ', '').upper()
    key = key.upper()
    
    n_cols = len(key)
    n_rows = (len(plaintext) + n_cols - 1) // n_cols  # ceiling division

    # Pad with 'X' if necessary
    padded_length = n_rows * n_cols
    padded_text = plaintext.ljust(padded_length, 'X')

    # Create the matrix row-wise
    matrix = [list(padded_text[i * n_cols:(i + 1) * n_cols]) for i in range(n_rows)]

    # Determine order of columns using sorted key
    order = sorted(range(n_cols), key=lambda k: key[k])

    # Read columns in the sorted order
    ciphertext = ''
    for col in order:
        for row in range(n_rows):
            ciphertext += matrix[row][col]

    return ciphertext


def columnar_decrypt(ciphertext, key):
    key = key.upper()
    n_cols = len(key)
    n_rows = (len(ciphertext) + n_cols - 1) // n_cols

    # Setup empty matrix
    matrix = [[''] * n_cols for _ in range(n_rows)]
    order = sorted(range(n_cols), key=lambda k: key[k])

    idx = 0
    # Fill columns according to sorted order
    for col in order:
        for row in range(n_rows):
            if idx < len(ciphertext):
                matrix[row][col] = ciphertext[idx]
                idx += 1

    # Read matrix row-wise for decrypted text
    plaintext = ''
    for row in matrix:
        plaintext += ''.join(row)

    # Remove padding 'X's
    return plaintext.rstrip('X')

message = "NETWORK SECURITY"

encrypted = columnar_encrypt(message, key)
print("Encrypted:", encrypted)

decrypted = columnar_decrypt(encrypted, key)
print("Decrypted:", decrypted)






# 2. Rail Fence Encryption


def rail_fence_encrypt(text, rails):
    # Create a list of empty strings for each rail
    fence = ['' for _ in range(rails)]
    
    rail = 0  # current rail index
    direction = 1  # 1 means moving down, -1 means moving up

    for char in text.replace(" ", ""):  # remove spaces
        fence[rail] += char  # add character to current rail
        
        # Change direction if we hit top or bottom rail
        if rail == 0:
            direction = 1
        elif rail == rails - 1:
            direction = -1
        
        # Move to the next rail
        rail += direction

    # Join all rails to get the ciphertext
    return ''.join(fence)


# Example usage
message = "Nizamuddin Tandel - 06"
ciphertext = rail_fence_encrypt(message, 3)
print("Plaintext: ", message)
print("Ciphertext:", ciphertext)




#3.Caeser Cipher


def caesar_encrypt(text, shift):
    encrypted = ""
    for char in text:
        if char.isalpha():
            # Determine the case and base ASCII value
            ascii_base = ord('A') if char.isupper() else ord('a')
            
            # Convert to 0–25 range, apply shift, and mod 26 to wrap around
            shifted = (ord(char) - ascii_base + shift) % 26 + ascii_base
            encrypted += chr(shifted)
        else:
            # Keep non-alphabetic characters unchanged
            encrypted += char
    return encrypted


def caesar_decrypt(text, shift):
    # Decrypt by shifting in the opposite direction
    return caesar_encrypt(text, -shift)


# Example usage
def main():
    message = "Hello, Nizam"
    shift = 3

    # Encrypt the message
    encrypted_message = caesar_encrypt(message, shift)
    print(f"Original message:  {message}")
    print(f"Encrypted message: {encrypted_message}")

    # Decrypt the message
    decrypted_message = caesar_decrypt(encrypted_message, shift)
    print(f"Decrypted message: {decrypted_message}")


if __name__ == "__main__":
    main()




#4.RSA Encryption and Decryption


import random

# Check if a number is prime
def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True


# Generate a random prime number within a range
def generate_prime(min_value, max_value):
    prime = random.randint(min_value, max_value)
    while not is_prime(prime):
        prime = random.randint(min_value, max_value)
    return prime


# Compute modular inverse of e under mod phi
def mod_inverse(e, phi):
    for d in range(1, phi):
        if (e * d) % phi == 1:
            return d
    return None


# Generate RSA key pair (public and private)
def generate_keypair():
    p = generate_prime(100, 1000)
    q = generate_prime(100, 1000)
    while p == q:
        q = generate_prime(100, 1000)

    n = p * q
    phi = (p - 1) * (q - 1)

    e = random.randint(2, phi - 1)
    while not is_prime(e):
        e = random.randint(2, phi - 1)

    d = mod_inverse(e, phi)
    if d is None:
        return generate_keypair()

    return (e, n), (d, n)


# Encrypt a message using the public key
def encrypt(public_key, plaintext):
    e, n = public_key
    cipher = [pow(ord(char), e, n) for char in plaintext]
    return cipher


# Decrypt a message using the private key
def decrypt(private_key, ciphertext):
    d, n = private_key
    plain = [chr(pow(char, d, n)) for char in ciphertext]
    return ''.join(plain)


# Main function for testing
if __name__ == '__main__':
    public_key, private_key = generate_keypair()
    print(f"Public key: {public_key}")
    print(f"Private key: {private_key}")

    message = "Hello, RSA!"
    print(f"\nOriginal message: {message}")

    encrypted_msg = encrypt(public_key, message)
    print(f"Encrypted message: {encrypted_msg}")

    decrypted_msg = decrypt(private_key, encrypted_msg)
    print(f"Decrypted message: {decrypted_msg}")




#5.Diffie-Hellman Key Exchange


import random

# Check if a number is prime
def is_prime(n):
    if n < 2:
        return False
    for i in range(2, int(n ** 0.5) + 1):
        if n % i == 0:
            return False
    return True


# Generate a random prime number between 100 and 1000
def generate_prime():
    primes = [i for i in range(100, 1000) if is_prime(i)]
    return random.choice(primes)


# Find a primitive root modulo p
def generate_primitive_root(p):
    for g in range(2, p):
        if all(pow(g, powers, p) != 1 for powers in range(1, p - 1)):
            return g
    return None


# Perform Diffie–Hellman key exchange
def diffie_hellman():
    # Publicly shared values
    p = generate_prime()
    g = generate_primitive_root(p)
    print(f"Publicly Shared Prime (p): {p}")
    print(f"Publicly Shared Base (g):  {g}")

    # Private keys (chosen secretly)
    alice_private = random.randint(1, p - 1)
    bob_private = random.randint(1, p - 1)
    print(f"\nAlice's Private Key: {alice_private}")
    print(f"Bob's Private Key:   {bob_private}")

    # Public keys (shared openly)
    alice_public = pow(g, alice_private, p)
    bob_public = pow(g, bob_private, p)
    print(f"\nAlice's Public Key: {alice_public}")
    print(f"Bob's Public Key:   {bob_public}")

    # Shared secrets (computed independently)
    alice_shared_secret = pow(bob_public, alice_private, p)
    bob_shared_secret = pow(alice_public, bob_private, p)

    print(f"\nAlice's Shared Secret: {alice_shared_secret}")
    print(f"Bob's Shared Secret:   {bob_shared_secret}")

    # Verify both parties got the same shared secret
    if alice_shared_secret == bob_shared_secret:
        print("\n Success: Shared secrets match!")
    else:
        print("\n Error: Shared secrets do not match!")


if __name__ == "__main__":
    diffie_hellman()



#6.MD5 Algorithm


import math

# MD5 constants
K = [
    int((1 << 32) * abs(math.sin(i + 1))) & 0xFFFFFFFF
    for i in range(64)
]

# Shift amounts (per round)
S = [
    7, 12, 17, 22,  7, 12, 17, 22,  7, 12, 17, 22,  7, 12, 17, 22,
    5,  9, 14, 20,  5,  9, 14, 20,  5,  9, 14, 20,  5,  9, 14, 20,
    4, 11, 16, 23,  4, 11, 16, 23,  4, 11, 16, 23,  4, 11, 16, 23,
    6, 10, 15, 21,  6, 10, 15, 21,  6, 10, 15, 21,  6, 10, 15, 21
]

def left_rotate(x, c):
    x &= 0xFFFFFFFF
    return ((x << c) | (x >> (32 - c))) & 0xFFFFFFFF

def md5(message: str) -> str:
    # Initialize variables (MD5 initial buffer values)
    a0 = 0x67452301
    b0 = 0xEFCDAB89
    c0 = 0x98BADCFE
    d0 = 0x10325476

    # Pre-processing: Convert message to bytes and pad it
    message_bytes = bytearray(message.encode('utf-8'))
    orig_len_in_bits = (8 * len(message_bytes)) & 0xFFFFFFFFFFFFFFFF

    # Append '1' bit (0x80), then pad with zeros until length ≡ 448 (mod 512)
    message_bytes.append(0x80)
    while len(message_bytes) % 64 != 56:
        message_bytes.append(0)

    # Append original length (in bits) as 64-bit little-endian
    message_bytes += orig_len_in_bits.to_bytes(8, byteorder='little')

    # Process message in 512-bit (64-byte) chunks
    for chunk_start in range(0, len(message_bytes), 64):
        chunk = message_bytes[chunk_start:chunk_start + 64]

        # Initialize hash value for this chunk
        A, B, C, D = a0, b0, c0, d0

        # Main loop: 64 steps divided into 4 rounds
        for i in range(64):
            if 0 <= i <= 15:
                F = (B & C) | (~B & D)
                g = i
            elif 16 <= i <= 31:
                F = (D & B) | (~D & C)
                g = (5 * i + 1) % 16
            elif 32 <= i <= 47:
                F = B ^ C ^ D
                g = (3 * i + 5) % 16
            else:
                F = C ^ (B | ~D)
                g = (7 * i) % 16

            # Add message word and constant
            F = (F + A + K[i] + int.from_bytes(chunk[4 * g:4 * g + 4], 'little')) & 0xFFFFFFFF

            # Rotate and update
            A, D, C, B = D, C, B, (B + left_rotate(F, S[i])) & 0xFFFFFFFF

        # Update hash values
        a0 = (a0 + A) & 0xFFFFFFFF
        b0 = (b0 + B) & 0xFFFFFFFF
        c0 = (c0 + C) & 0xFFFFFFFF
        d0 = (d0 + D) & 0xFFFFFFFF

    # Combine final hash and convert to hex
    digest = (
        a0.to_bytes(4, 'little') +
        b0.to_bytes(4, 'little') +
        c0.to_bytes(4, 'little') +
        d0.to_bytes(4, 'little')
    )
    return digest.hex()

    ]

        print(f"MD5('{s}') = {md5(s)}")

import hashlib

def hash_string_md5(input_string: str) -> str:
    """
    Calculates the MD5 hash of a given string and returns the hex digest.
    """
    md5_hash = hashlib.md5()                 # Create an MD5 hash object
    md5_hash.update(input_string.encode('utf-8'))  # Feed bytes into the hasher
    return md5_hash.hexdigest()              # Hexadecimal representation

# Example usage
if __name__ == "__main__":
    text_to_hash = "Hello, world!"
    md5_result = hash_string_md5(text_to_hash)
    print(f"The MD5 hash of '{text_to_hash}' is: {md5_result}")



#7.SHA-256

from decimal import Decimal, getcontext
import hashlib

# High precision for constant generation
getcontext().prec = 100

def generate_primes(n: int):
    primes = []
    num = 2
    while len(primes) < n:
        if all(num % p != 0 for p in primes):
            primes.append(num)
        num += 1
    return primes

def cbrt_decimal(x: Decimal) -> Decimal:
    # Newton's method for cube root
    if x == 0:
        return Decimal(0)
    # Initial guess
    g = x if x < 1 else x / 3
    for _ in range(50):
        g = (Decimal(2) * g + x / (g * g)) / Decimal(3)
    return g

def frac32_of_root(n: int, cube: bool = False) -> int:
    x = Decimal(n)
    r = cbrt_decimal(x) if cube else x.sqrt()
    frac = r - int(r)  # fractional part
    # floor(frac * 2^32) & 0xffffffff
    return int(frac * (1 << 32)) & 0xFFFFFFFF

def generate_sha256_constants():
    # Initial hash values: frac parts of sqrt(primes 2..19)
    primes_h = generate_primes(8)
    H = [frac32_of_root(p, cube=False) for p in primes_h]

    # Round constants: frac parts of cbrt(primes 2..311)
    primes_k = generate_primes(64)
    K = [frac32_of_root(p, cube=True) for p in primes_k]

    return H, K

def right_rotate(x: int, n: int) -> int:
    return ((x >> n) | (x << (32 - n))) & 0xFFFFFFFF

def sha256(message: str) -> str:
    H, K = generate_sha256_constants()

    # Preprocess
    m = bytearray(message.encode('utf-8'))
    bit_len = (len(m) * 8) & 0xFFFFFFFFFFFFFFFF
    m.append(0x80)
    while (len(m) * 8) % 512 != 448:
        m.append(0x00)
    m += bit_len.to_bytes(8, 'big')

    # Process each 512-bit chunk
    for i in range(0, len(m), 64):
        chunk = m[i:i+64]
        w = [0] * 64

        for j in range(16):
            w[j] = int.from_bytes(chunk[j*4:j*4+4], 'big')

        for j in range(16, 64):
            s0 = right_rotate(w[j-15], 7) ^ right_rotate(w[j-15], 18) ^ (w[j-15] >> 3)
            s1 = right_rotate(w[j-2], 17) ^ right_rotate(w[j-2], 19) ^ (w[j-2] >> 10)
            w[j] = (w[j-16] + s0 + w[j-7] + s1) & 0xFFFFFFFF

        a, b, c, d, e, f, g, h_ = H

        for j in range(64):
            S1 = right_rotate(e, 6) ^ right_rotate(e, 11) ^ right_rotate(e, 25)
            ch = (e & f) ^ ((~e & 0xFFFFFFFF) & g)
            t1 = (h_ + S1 + ch + K[j] + w[j]) & 0xFFFFFFFF
            S0 = right_rotate(a, 2) ^ right_rotate(a, 13) ^ right_rotate(a, 22)
            maj = (a & b) ^ (a & c) ^ (b & c)
            t2 = (S0 + maj) & 0xFFFFFFFF

            h_ = g
            g = f
            f = e
            e = (d + t1) & 0xFFFFFFFF
            d = c
            c = b
            b = a
            a = (t1 + t2) & 0xFFFFFFFF

        H[0] = (H[0] + a) & 0xFFFFFFFF
        H[1] = (H[1] + b) & 0xFFFFFFFF
        H[2] = (H[2] + c) & 0xFFFFFFFF
        H[3] = (H[3] + d) & 0xFFFFFFFF
        H[4] = (H[4] + e) & 0xFFFFFFFF
        H[5] = (H[5] + f) & 0xFFFFFFFF
        H[6] = (H[6] + g) & 0xFFFFFFFF
        H[7] = (H[7] + h_) & 0xFFFFFFFF

    return ''.join(f'{x:08x}' for x in H)

if __name__ == "__main__":
    msg = "hello world"
    print("sha256:", sha256(msg))
    # Quick verification against hashlib
    print("hashlib:", hashlib.sha256(msg.encode()).hexdigest())
